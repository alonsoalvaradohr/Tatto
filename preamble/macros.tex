% --- Virtual Oscilloscope Instrumentation Macro ---
% Syntax: \renderScope{x_pos}{y_pos}{width}{height}{title}{testpoint_label}{trace_code}
\newcommand{\renderScope}[7]{
    \begin{scope}[shift={(#1,#2)}]
        % Outer Frame
        \draw[scope_frame] ({-0.5*#3}, {-0.5*#4}) rectangle ({0.5*#3}, {0.5*#4});
        
        % Screen Area
        \filldraw[scope_screen] ({-0.5*#3+0.15}, {-0.5*#4+0.15}) rectangle ({0.5*#3-0.15}, {0.5*#4-0.15});
        
        % Internal Grid Graticule
        \draw[scope_grid] ({-0.5*#3+0.2}, {-0.5*#4+0.2}) grid ({0.5*#3-0.2}, {0.5*#4-0.2});
        
        % Major Axes
        \draw[scope_axis] ({-0.5*#3+0.2}, 0) -- ({0.5*#3-0.2}, 0);
        \draw[scope_axis] (0, {-0.5*#4+0.2}) -- (0, {0.5*#4-0.2});
        
        % Title & Channel Label
        \node[scope_annotation, anchor=north west] at ({-0.5*#3+0.25}, {0.5*#4-0.2}) {#5};
        \node[scope_annotation, anchor=north east, text=red!70!black] at ({0.5*#3-0.25}, {0.5*#4-0.2}) {#6};
        
        % Clipped Waveform Trace
        \begin{scope}
            \clip ({-0.5*#3+0.2}, {-0.5*#4+0.2}) rectangle ({0.5*#3-0.2}, {0.5*#4-0.2});
            #7
        \end{scope}
    \end{scope}
}
