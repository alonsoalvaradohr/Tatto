\begin{tikzpicture}[every node/.style={font=\sffamily\small}]
  % Rails: VRAW y=10; VOUT y=4; GND y=0.
  \draw (3,10) -- (18,10) node[right] {VRAW};
  \draw (3,0) -- (18,0);
  \draw (18,0) node[ground] {};
  \node[right] at (18,0) {GND};

  % NPN pass device: collector=VRAW, base=DRV, emitter=SENSE.
  \node[npn] (Q8) at (5,6) {};
  \node[right] at (5.8,6.55) {Q8};
  \draw (Q8.C) |- (5,10);
  \draw (5,10) node[circ,fill=black,inner sep=1pt] {};
  \draw (Q8.B) -- ++(-1,0) node[left] {DRV};

  % Current-sense resistor: Q8 emitter -> SENSE -> RSC -> VOUT.
  \coordinate (SENSEnode) at (5,4);
  \draw (Q8.E) -- (SENSEnode) to[R,l=RSC] (10,4);
  \draw (SENSEnode) node[circ,fill=black,inner sep=1pt] {};
  \node[below] at (SENSEnode) {SENSE};
  \draw (10,4) -- (17,4);
  \draw (10,4) node[circ,fill=black,inner sep=1pt] {};
  \node[above] at (10,4) {VOUT};

  % Current-limit transistor: base=SENSE, emitter=VOUT, collector=DRV.
  \node[npn] (Q9) at (13,6) {};
  \node[right] at (13.8,5.05) {Q9};
  \draw (Q9.B) -- ++(-1,0) node[left] {SENSE};
  \draw (Q9.E) |- (13,4);
  \draw (13,4) node[circ,fill=black,inner sep=1pt] {};
  \draw (Q9.C) -- ++(1,0) node[right] {DRV};

  % Divider: VOUT -> R1 -> FB -> R2 -> GND.
  \coordinate (FBnode) at (17,2);
  \draw (17,4) to[R,l=R1] (FBnode) to[R,l=R2] (17,0);
  \draw (17,4) node[circ,fill=black,inner sep=1pt] {};
  \draw (FBnode) node[circ,fill=black,inner sep=1pt] {};
  \node[right] at (FBnode) {FB};
